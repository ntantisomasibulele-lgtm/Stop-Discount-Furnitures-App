  <!-- FULL INTERACTIVE JAVASCRIPT LOGIC -->
  <script>
    const micButton = document.getElementById('micButton');
    const statusText = document.getElementById('statusText');
    const transcriptInput = document.getElementById('transcriptInput');
    const workflowSelect = document.getElementById('workflowSelect');
    const exportButton = document.getElementById('exportButton');
    const tableBody = document.getElementById('tableBody');
    const entryBadge = document.getElementById('entryBadge');
    const toastNotification = document.getElementById('toastNotification');
    const accentSaEnglishBtn = document.getElementById('accentSaEnglishBtn');
    const accentIsiXhosaBtn = document.getElementById('accentIsiXhosaBtn');
    const quickCopyBtn = document.getElementById('quickCopyBtn');
    const clearTextBtn = document.getElementById('clearTextBtn');

    const TRANSCRIPT_SA_ENGLISH =
      "Stock check complete for showroom floor. Two luxury leather couches checked in. One minor scratch reported on the left armrest of the unit. Camagu.";

    const TRANSCRIPT_ISIXHOSA =
      "I-odolo entsha yesofa. Custom 3-seater corner couch requested in dark gray suede fabric. Umthengi ubhatale idipozithi. Camagu.";

    let activeAccentProfile = 'SA English';
    let isRecording = false;
    let exportCount = 0;
    let toastTimer = null;
    let copyButtonTimer = null;
    const exportedRecords = [];

    // Accent Optimization Profile Toggle Logic
    function setAccentProfile(profile) {
      activeAccentProfile = profile;
      const activeClasses =
        'px-3.5 py-1 rounded-full text-xs font-bold transition cursor-pointer bg-amber-50 text-amber-900 border border-amber-500 shadow-2xs';
      const inactiveClasses =
        'px-3.5 py-1 rounded-full text-xs font-semibold transition cursor-pointer text-slate-600 border border-transparent hover:text-slate-900';

      if (profile === 'SA English') {
        accentSaEnglishBtn.className = activeClasses;
        accentIsiXhosaBtn.className = inactiveClasses;
      } else {
        accentIsiXhosaBtn.className = activeClasses;
        accentSaEnglishBtn.className = inactiveClasses;
      }
    }

    accentSaEnglishBtn.addEventListener('click', function () {
      setAccentProfile('SA English');
    });

    accentIsiXhosaBtn.addEventListener('click', function () {
      setAccentProfile('isiXhosa');
    });

    function getActiveSimulatedTranscript() {
      return activeAccentProfile === 'isiXhosa'
        ? TRANSCRIPT_ISIXHOSA
        : TRANSCRIPT_SA_ENGLISH;
    }

    // Clipboard Copy Button Logic
    quickCopyBtn.addEventListener('click', function () {
      const textToCopy = transcriptInput.value;

      if (navigator.clipboard && navigator.clipboard.writeText) {
        navigator.clipboard.writeText(textToCopy).catch(function () {
          transcriptInput.select();
          document.execCommand('copy');
        });
      } else {
        transcriptInput.select();
        document.execCommand('copy');
      }

      quickCopyBtn.textContent = 'Copied! ✓';
      quickCopyBtn.classList.add('bg-emerald-50', 'text-emerald-700', 'border-emerald-300');

      if (copyButtonTimer) {
        clearTimeout(copyButtonTimer);
      }
      copyButtonTimer = setTimeout(function () {
        quickCopyBtn.textContent = '📋 Copy';
        quickCopyBtn.classList.remove('bg-emerald-50', 'text-emerald-700', 'border-emerald-300');
      }, 2000);
    });

    // Clear Textarea Button Logic
    clearTextBtn.addEventListener('click', function () {
      transcriptInput.value = '';
      statusText.classList.remove('flashing-recording-text');
      statusText.textContent = 'Tap microphone to start 4-second field voice capture';
    });

    // Excel-Compatible CSV Formatter & Auto-Downloader
    function toCsvField(value) {
      return '"' + String(value).replace(/"/g, '""') + '"';
    }

    function triggerCsvAutoDownload(records) {
      const header = 'Date/Time,Workflow Type,Logged Details\r\n';
      const rows = records.map(function (item) {
        return (
          toCsvField(item.dateTime) +
          ',' +
          toCsvField(item.workflowType) +
          ',' +
          toCsvField(item.loggedDetails)
        );
      });

      const csvContent = '\uFEFF' + header + rows.join('\r\n');
      const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
      const downloadUrl = URL.createObjectURL(blob);

      const downloadLink = document.createElement('a');
      downloadLink.href = downloadUrl;
      downloadLink.setAttribute('download', 'Discount_Furnishers_Floor_Log.csv');
      document.body.appendChild(downloadLink);
      downloadLink.click();
      document.body.removeChild(downloadLink);

      setTimeout(function () {
        URL.revokeObjectURL(downloadUrl);
      }, 1000);
    }

    // 4-Second Microphone Simulation Timer
    micButton.addEventListener('click', function () {
      if (isRecording) return;
      isRecording = true;

      micButton.classList.remove('bg-amber-600', 'hover:bg-amber-700');
      micButton.classList.add('recording-active');

      statusText.textContent = 'Recording field audio...';
      statusText.classList.add('flashing-recording-text');

      setTimeout(function () {
        isRecording = false;

        micButton.classList.remove('recording-active');
        micButton.classList.add('bg-amber-600', 'hover:bg-amber-700');

        statusText.classList.remove('flashing-recording-text');
        statusText.textContent = 'Audio capture complete — ready to export.';
        transcriptInput.value = getActiveSimulatedTranscript();
      }, 4000);
    });

    // Export Button: Prepend Table Row + Increment Counter + Toast Alert + Download CSV
    exportButton.addEventListener('click', function () {
      let detailsText = transcriptInput.value.trim();

      if (!detailsText) {
        detailsText = getActiveSimulatedTranscript();
        transcriptInput.value = detailsText;
      }

      const emptyRow = document.getElementById('emptyRow');
      if (emptyRow) {
        emptyRow.remove();
      }

      const now = new Date();
      const dateTimeString = now.toLocaleString('en-ZA', {
        year: 'numeric',
        month: 'short',
        day: '2-digit',
        hour: '2-digit',
        minute: '2-digit',
        second: '2-digit',
        hour12: false
      });
      const selectedWorkflow = workflowSelect.value;

      const newRow = document.createElement('tr');
      newRow.className = 'hover:bg-amber-50/40 transition-colors';

      const dateCell = document.createElement('td');
      dateCell.className = 'py-3.5 px-4 font-mono text-xs text-slate-600 whitespace-nowrap align-top';
      dateCell.textContent = dateTimeString;

      const workflowCell = document.createElement('td');
      workflowCell.className = 'py-3.5 px-4 align-top';
      const workflowBadge = document.createElement('span');
      workflowBadge.className = 'inline-block px-2.5 py-1 rounded-md text-xs font-bold bg-amber-100 text-amber-900 border border-amber-200 whitespace-nowrap';
      workflowBadge.textContent = selectedWorkflow;
      workflowCell.appendChild(workflowBadge);

      const detailsCell = document.createElement('td');
      detailsCell.className = 'py-3.5 px-4 text-slate-800 leading-relaxed align-top';
      detailsCell.textContent = detailsText;

      newRow.appendChild(dateCell);
      newRow.appendChild(workflowCell);
      newRow.appendChild(detailsCell);

      tableBody.prepend(newRow);

      exportCount += 1;
      entryBadge.textContent = exportCount === 1 ? '1 Entry' : exportCount + ' Entries';

      exportedRecords.unshift({
        dateTime: dateTimeString,
        workflowType: selectedWorkflow,
        loggedDetails: detailsText
      });
      triggerCsvAutoDownload(exportedRecords);

      toastNotification.classList.remove('opacity-0');
      toastNotification.classList.add('opacity-100');

      if (toastTimer) {
        clearTimeout(toastTimer);
      }
      toastTimer = setTimeout(function () {
        toastNotification.classList.remove('opacity-100');
        toastNotification.classList.add('opacity-0');
      }, 3000);
    });
  </script>
